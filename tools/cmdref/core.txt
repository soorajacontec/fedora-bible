@ls ~~ List directory contents
$ ls [OPTION]... [FILE]...
- -l ~~ Long format: permissions, owner, size, date
- -a ~~ Show hidden files (starting with .)
- -A ~~ Hidden files, but not . and ..
- -h ~~ Human-readable sizes (K, M, G) with -l
- -t ~~ Sort by modification time, newest first
- -S ~~ Sort by size, largest first
- -r ~~ Reverse the sort order
- -R ~~ List subdirectories recursively
- -d ~~ List the directory itself, not its contents
- -1 ~~ One entry per line
- -i ~~ Show inode numbers
- -Z ~~ Show SELinux security context
- --color=auto ~~ Colour by file type
- --group-directories-first ~~ Folders before files
> ls -lah ~~ Everything, long format, human sizes
> ls -lt | head ~~ 10 most recently changed
> ls -lS --block-size=M ~~ Biggest files first, in MB
> ls -d */ ~~ Only directories
> ls -Z /var/www ~~ SELinux labels

@cd ~~ Change the current directory
$ cd [DIR]
- (no args) ~~ Go to your home directory
- - ~~ Go back to the previous directory
- .. ~~ Parent directory
- ~user ~~ Another user's home
- -P ~~ Resolve symlinks to the real path
> cd ~/Downloads ~~ Into Downloads
> cd - ~~ Toggle back
> cd ../.. ~~ Up two levels

@pwd ~~ Print the current directory
$ pwd [-L|-P]
- -L ~~ Logical path (keeps symlinks, default)
- -P ~~ Physical path (symlinks resolved)
> pwd ~~ Where am I?

@cp ~~ Copy files and directories
$ cp [OPTION]... SOURCE DEST
$ cp [OPTION]... SOURCE... DIRECTORY
- -r ~~ Copy directories recursively
- -a ~~ Archive: recursive + keep permissions, times, links
- -i ~~ Ask before overwriting
- -n ~~ Never overwrite
- -u ~~ Copy only when source is newer
- -v ~~ Verbose: show each file
- -p ~~ Preserve mode, owner, timestamps
- -l ~~ Hard-link instead of copying
- --reflink=auto ~~ Instant copy-on-write clone (Btrfs, XFS)
- -t DIR ~~ Target directory first (good with xargs)
> cp file.txt backup.txt ~~ Simple copy
> cp -a ~/Documents /run/media/$USER/USB/ ~~ Copy a folder, keep everything
> cp -iv *.jpg ~/Pictures/ ~~ Ask before overwriting, show progress
> cp --reflink=auto big.iso copy.iso ~~ Instant copy on Btrfs

@mv ~~ Move or rename files
$ mv [OPTION]... SOURCE DEST
$ mv [OPTION]... SOURCE... DIRECTORY
- -i ~~ Ask before overwriting
- -n ~~ Never overwrite
- -u ~~ Move only when source is newer
- -v ~~ Show what is moved
- -b ~~ Back up files that would be overwritten
- -t DIR ~~ Target directory first
> mv old.txt new.txt ~~ Rename
> mv -i *.pdf ~/Documents/ ~~ Move, ask before overwriting
> mv -v report{,.bak} ~~ report → report.bak (brace expansion)

- -I ~~ Ask once before removing more than 3 files
- -v ~~ Show each removed file
- -d ~~ Remove empty directories
- --one-file-system ~~ Don't cross into other mounted filesystems

> rm -ri old-project/ ~~ Delete a folder, confirming each file
> rm -rf build/ ~~ Delete a folder, no questions (careful!)
> find . -name "*.tmp" -delete ~~ Safer bulk delete by pattern

@find ~~ Search for files in a directory tree
$ find [PATH...] [TESTS] [ACTIONS]
- -name PATTERN ~~ File name matches (quote wildcards!)
- -iname PATTERN ~~ Case-insensitive name
- -type f|d|l ~~ Files / directories / symlinks
- -size +100M ~~ Larger than 100 MB (-size -1k smaller)
- -mtime -7 ~~ Modified in the last 7 days (+30 = older)
- -mmin -60 ~~ Modified in the last 60 minutes
- -newer FILE ~~ Newer than FILE
- -user NAME ~~ Owned by user
- -perm -4000 ~~ Has the setuid bit
- -empty ~~ Empty files or directories
- -maxdepth N ~~ Don't go deeper than N levels
- -not / ! ~~ Negate the next test
- -exec CMD {} + ~~ Run a command on the results
- -delete ~~ Delete matches (put it last!)
- -print0 ~~ NUL-separated output for xargs -0
> find ~ -name "*.pdf" ~~ All PDFs in your home
> find / -xdev -size +1G 2>/dev/null ~~ Files over 1 GB on this filesystem
> find . -type f -mtime -1 ~~ Changed in the last day
> find . -name "*.log" -mtime +30 -delete ~~ Delete logs older than 30 days
> find . -type f -exec chmod 644 {} + ~~ Fix file permissions in bulk

@grep ~~ Search text for patterns
$ grep [OPTION]... PATTERN [FILE]...
- -i ~~ Ignore case
- -r ~~ Search folders recursively
- -n ~~ Show line numbers
- -v ~~ Invert: lines that do NOT match
- -w ~~ Whole words only
- -l ~~ Only file names that match
- -c ~~ Count matching lines
- -E ~~ Extended regex (a|b, +, ?)
- -F ~~ Fixed string, no regex
- -o ~~ Print only the matched part
- -A N / -B N / -C N ~~ N lines after / before / around matches
- --include=GLOB ~~ Only search matching files
- --exclude-dir=DIR ~~ Skip folders
- -q ~~ Quiet: just the exit status (scripts)

> grep -i error /var/log/messages ~~ Case-insensitive
> grep -E "fail|error" app.log ~~ Either word
> ps aux | grep -v grep | grep nginx ~~ Filter command output
> grep -rl --include="*.conf" "Listen" /etc ~~ Which config files mention it

@sed ~~ Stream editor: find/replace and edit text
$ sed [OPTION]... 'SCRIPT' [FILE]...
- -i ~~ Edit the file in place
- -i.bak ~~ In place, keep a .bak copy
- -n ~~ Print only lines you ask for (with p)
- -E ~~ Extended regex
- s/old/new/g ~~ Replace all matches on each line
- Nd ~~ Delete line N (e.g. 1d)
- /re/d ~~ Delete lines matching re
- N,Mp ~~ Print lines N to M (with -n)

@awk ~~ Pattern scanning and column processing
$ awk [-F SEP] 'PROGRAM' [FILE]...
- -F SEP ~~ Field separator (default: spaces/tabs)
- $1, $2 … ~~ Fields (columns); $0 = whole line
- NR ~~ Current line number
- NF ~~ Number of fields; $NF = last field
- BEGIN{} / END{} ~~ Run before / after all lines
- -v var=value ~~ Pass a variable in
> awk '{print $1}' access.log ~~ First column
> awk -F: '{print $1, $7}' /etc/passwd ~~ Users and their shells
> awk '$3 > 100' data.txt ~~ Rows where column 3 > 100
> awk '{sum += $2} END {print sum}' sales.txt ~~ Total of a column
> df -h | awk 'NR>1 && $5+0 > 80' ~~ Filesystems over 80% full

$ man [SECTION] NAME
$ man -k KEYWORD
- -k WORD ~~ Search page titles (same as apropos)
- -f NAME ~~ Short description (whatis)
- 1 / 5 / 8 ~~ Sections: commands / config files / admin
- /word ~~ (inside) search; q quits
> man ls ~~ Manual for ls
> man 5 crontab ~~ The crontab file format, not the command
> man -k firewall ~~ Find related commands

@history ~~ Shell command history
$ history [N]
- N ~~ Last N commands
- -c ~~ Clear history
- -d N ~~ Delete entry N
- !N ~~ Re-run command number N
- !! ~~ Re-run the last command (e.g. sudo !!)
- Ctrl+R ~~ Search history interactively
> history | grep dnf ~~ Find a previous command
> sudo !! ~~ Repeat the last command with sudo
