@for ~~ Bash loop over a list
$ for VAR in LIST; do COMMANDS; done
$ for ((i=1; i<=N; i++)); do COMMANDS; done
- in a b c ~~ Loop over words
- in *.jpg ~~ Loop over files (quote "$f"!)
- in {1..10} ~~ Number range
- in $(cmd) ~~ Over command output (careful with spaces)
- break / continue ~~ Leave / skip an iteration
> for f in *.png; do magick "$f" "${f%.png}.webp"; done ~~ Convert every PNG
> for h in web1 web2 web3; do ssh "$h" uptime; done ~~ Same command on several servers
> for i in {1..5}; do echo "Try $i"; done ~~ Count

@export ~~ Make a shell variable visible to programs
$ export NAME=VALUE
- export NAME ~~ Export an existing variable
- -n NAME ~~ Un-export
- -p ~~ List exported variables
- ~/.bashrc ~~ Put exports here to keep them
> export EDITOR=nano ~~ Default editor for this session
> export PATH="$HOME/.local/bin:$PATH" ~~ Add a folder to PATH

@shopt ~~ Bash shell options
$ shopt [-s|-u] OPTION
- -s ~~ Set (enable)
- -u ~~ Unset (disable)
- globstar ~~ ** matches folders recursively
- extglob ~~ Extended patterns like !(*.txt)
- nullglob ~~ Unmatched globs expand to nothing
- dotglob ~~ * also matches hidden files
> shopt -s globstar && ls **/*.md ~~ All Markdown files below here

> udisksctl mount -b /dev/sdb1 ~~ Mount a USB stick
> udisksctl power-off -b /dev/sdb ~~ Safe to unplug
